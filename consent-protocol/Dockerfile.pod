# syntax=docker/dockerfile:1
# Slim pod image — runs ONLY the sovereign per-user agent + storage/enforcement
# surface (`pod_server:app`), NOT the consent control plane. See pod_server.py.
#
# It shares the base build with the hub image; what differs is the RUNTIME surface
# (the entrypoint mounts an allowlist of routers) and pod-mode (no fleet workers).
# The pod and hub intentionally share the locked application dependency graph, but
# the graph is installed in build time project mode so its explicit CPU-only Torch
# source is honored. The runtime stage receives only the built environment and app
# source; it never installs dependencies or reaches a package registry.

# Stage 1: build dependencies
FROM python:3.13-slim as builder
WORKDIR /app
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1
RUN apt-get update && apt-get install -y --no-install-recommends gcc \
    && rm -rf /var/lib/apt/lists/*
RUN pip install uv
COPY pyproject.toml uv.lock ./
RUN uv venv /opt/venv
ENV VIRTUAL_ENV=/opt/venv \
    PATH=/opt/venv/bin:$PATH
RUN --mount=type=cache,target=/root/.cache/uv \
    uv sync --frozen --no-dev --no-install-project --active --compile-bytecode
# Pin Browser dependencies independently. Chromium starts only on admitted demand
# inside the same owner container; the dependency tree is not a trust boundary.
COPY browser_runtime/requirements.lock /tmp/browser-requirements.lock
RUN uv venv /opt/browser/venv \
    && uv pip sync --python /opt/browser/venv/bin/python --require-hashes /tmp/browser-requirements.lock

# Private command retrieval and Drive use the same pinned local embedding
# model as the hub. Fetch and exercise it at build time, never on an owner turn.
COPY hushh_mcp/services/embedding_client_leaf.py hushh_mcp/services/embedding_client_leaf.py
COPY scripts/ops/bake_drive_embedding_model.py scripts/ops/bake_drive_embedding_model.py
RUN python -m scripts.ops.bake_drive_embedding_model

# Stage 2: production runtime
FROM python:3.13-slim
# The build tag, baked in so the running process can say which build it is
# without asking anyone (`/pod/info` and the heartbeat report it). Empty when
# built outside Cloud Build; the pod then reports nothing rather than a guess.
ARG POD_IMAGE_TAG=""
WORKDIR /app
COPY --from=builder /opt/venv /opt/venv
COPY --from=builder /opt/hushh/models /opt/hushh/models
COPY --from=builder /opt/browser/venv /opt/browser/venv
RUN PLAYWRIGHT_BROWSERS_PATH=/opt/browser/browsers \
       /opt/browser/venv/bin/playwright install --with-deps --only-shell chromium \
    && rm -rf /var/lib/apt/lists/*
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    VIRTUAL_ENV=/opt/venv \

    HUSSH_POD_MODE=1 \

COPY . .

# output with the runtime reader and fails if the catalog is absent or invalid.
RUN HF_HUB_OFFLINE=1 python -m scripts.ops.bake_action_catalog_vectors
RUN python -m compileall -q .
# Fixed Browser worker imports; these paths do not isolate it from One.
COPY hushh_mcp/services/pod_browser/launch_protocol.py \
     hushh_mcp/services/pod_browser/launch_service.py \
     hushh_mcp/services/pod_browser/container_worker.py \

     hushh_mcp/services/pod_browser/network.py \
     hushh_mcp/services/pod_browser/downloads.py \
     hushh_mcp/services/pod_browser/contracts.py \
     hushh_mcp/services/pod_browser/mailbox.py \
     hushh_mcp/services/pod_browser/playwright_executor.py \
     hushh_mcp/services/pod_browser/session_state.py \
     hushh_mcp/services/pod_browser/scratch.py \
     hushh_mcp/services/pod_browser/public_profile.py \
     hushh_mcp/services/pod_browser/origin.py /opt/browser/source/hushh_mcp/services/pod_browser/
COPY hushh_mcp/services/public_endpoint_policy.py \
     hushh_mcp/services/public_network_address.py \
     hushh_mcp/services/mcp_public_http.py /opt/browser/source/hushh_mcp/services/
COPY browser_runtime/pod_entrypoint.py browser_runtime/gunicorn_lifecycle.py browser_runtime/owner_entrypoint.py /opt/browser/
RUN /opt/browser/venv/bin/python -I -c "import sys; sys.path.insert(0, '/opt/browser/source'); from hushh_mcp.services.pod_browser.scratch import MemoryScratch; from hushh_mcp.services.pod_browser.playwright_executor import PlaywrightExecutor; from hushh_mcp.services.pod_browser.network import BrowserNetworkBroker; from hushh_mcp.services.pod_browser.container_worker import main"
# The fixed bootstrap drops privileges before One or Browser imports.
RUN groupadd --gid 10001 agent && useradd --system --uid 10001 --gid 10001 --home-dir /app appuser
# gunicorn's control server writes a runtime dir under CWD/HOME; the code tree is
# root-owned + read-only for the runtime user, so give it one writable path.
RUN mkdir -p /app/.gunicorn && chown 10001:10001 /app/.gunicorn && chmod 0700 /app/.gunicorn
EXPOSE 8080
# Slim pod entrypoint: agent + storage + enforcement only (no consent control plane).
#
# ONE worker, for CORRECTNESS, not for cost.
#
# `GcpBackend` pins maxScale=1 and says why in its own comment: the pod's storage
# engine assumes a single writer per pod (SQLite `BEGIN IMMEDIATE` over one file,
# with the sealed commit log as the system of record). That invariant was enforced
# at the Cloud Run layer and then violated one process layer down -- this line ran
# `-w 2`, and gunicorn does not preload by default, so each worker imported the app
# independently, ran `@app.on_event("startup")` independently, and got its OWN
# module-level `_STATE` in `pod_self_registration`.
#
# The consequence is not a performance nuance. A pod generates an ephemeral X25519
# keypair at boot, so two workers meant TWO IDENTITIES INSIDE ONE POD, and
# `/pod/public-key` returned whichever worker happened to answer the hub's
# collection call. Anything the hub then sealed to that key had a coin-flip chance
# of being unreadable by the worker that served the next turn. Two heartbeat loops
# ran for the same reason.
#
# `--preload` would share the parent's imports but not fix it: post-fork state
# still diverges, and the storage engine would still have two writers. One worker
# is the honest expression of the invariant the rest of the system already assumes.
ENTRYPOINT ["/usr/local/bin/python", "-I", "-S", "/opt/browser/pod_entrypoint.py"]
